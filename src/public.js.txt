// Public page: draws only what /api/public returns (the server enforces visibility).
(function () {
  const $ = (s) => document.querySelector(s);
  const fmt = new Intl.NumberFormat('en-US');
  const day = (iso, opts) => new Date(iso + 'T00:00:00Z').toLocaleDateString('en-US', { timeZone: 'UTC', ...opts });
  const el = (tag, attrs = {}, text) => { const n = document.createElement(tag); for (const [k, v] of Object.entries(attrs)) n.setAttribute(k, v); if (text !== undefined) n.textContent = text; return n; };
  const NS = 'http://www.w3.org/2000/svg';
  const svgEl = (tag, attrs) => { const n = document.createElementNS(NS, tag); for (const [k, v] of Object.entries(attrs)) n.setAttribute(k, v); return n; };

  function totals(t) {
    const set = (k, v) => { const n = document.querySelector(`[data-k="${k}"]`); if (n) n.textContent = v; };
    set('total', fmt.format(t.total)); set('month', fmt.format(t.month)); set('week', fmt.format(t.week)); set('latest_day', fmt.format(t.latest_day));
    set('since', t.first_date ? 'Since ' + day(t.first_date, { month: 'short', day: 'numeric' }) : '');
    if (t.latest_date) {
      set('month_label', day(t.latest_date, { month: 'long', year: 'numeric' }));
      set('week_label', 'Week of ' + day(t.week_start, { month: 'short', day: 'numeric' }));
      set('latest_label', day(t.latest_date, { weekday: 'short', month: 'short', day: 'numeric' }));
    }
    $('#totals').hidden = false;
  }

  function dailyChart(rows) {
    const box = $('#daily-chart');
    const draw = () => {
      box.replaceChildren();
      const W = Math.max(300, box.clientWidth), H = 230, m = { t: 20, r: 8, b: 26, l: 34 };
      const n = rows.length || 1;
      const max = Math.max(1, ...rows.map((r) => r.auto + r.manual));
      const raw = max / 4, mag = Math.pow(10, Math.floor(Math.log10(raw))); const step = [1, 2, 2.5, 5, 10].map((f) => f * mag).find((s) => s >= raw); const top = step * 4;
      const y = (v) => H - m.b - (v / top) * (H - m.t - m.b);
      const bw = (W - m.l - m.r) / n; const gap = Math.min(16, bw * 0.3);
      const svg = svgEl('svg', { viewBox: `0 0 ${W} ${H}`, role: 'img', 'aria-label': 'Applications per day, split into automatic and manual submissions' });
      for (let i = 0; i <= 4; i++) {
        const v = (top / 4) * i; const yy = y(v);
        svg.append(svgEl('line', { x1: m.l, x2: W - m.r, y1: yy, y2: yy, class: 'gridline' }));
        const t = svgEl('text', { x: m.l - 6, y: yy + 4, 'text-anchor': 'end' }); t.textContent = Math.round(v); svg.append(t);
      }
      const tip = el('div', { class: 'tip' });
      const labelEvery = Math.ceil(n / Math.max(1, Math.floor((W - m.l) / 64)));
      rows.forEach((r, i) => {
        const x = m.l + i * bw + gap / 2, w = Math.max(2, bw - gap);
        const g = svgEl('g', { tabindex: 0 });
        g.append(svgEl('rect', { x, y: y(r.auto), width: w, height: H - m.b - y(r.auto), fill: 'var(--accent)', rx: 2 }));
        g.append(svgEl('rect', { x, y: y(r.auto + r.manual), width: w, height: y(r.auto) - y(r.auto + r.manual), fill: 'var(--accent-2)', rx: 2 }));
        const total = svgEl('text', { x: x + w / 2, y: y(r.auto + r.manual) - 5, 'text-anchor': 'middle' }); total.textContent = r.auto + r.manual; total.style.fill = 'var(--ink)';
        if (bw > 22) g.append(total);
        if (i % labelEvery === 0) { const lb = svgEl('text', { x: x + w / 2, y: H - 8, 'text-anchor': 'middle' }); lb.textContent = day(r.date, { month: 'short', day: 'numeric' }); svg.append(lb); }
        const show = (ev) => {
          tip.textContent = `${day(r.date, { weekday: 'short', month: 'short', day: 'numeric' })}: ${r.auto + r.manual} (${r.auto} automatic, ${r.manual} by Chris)`;
          const bx = box.getBoundingClientRect(); const px = (ev.clientX || bx.left + x) - bx.left;
          tip.style.left = Math.min(Math.max(0, px - 80), bx.width - 230) + 'px'; tip.style.top = '0px'; tip.style.opacity = 1;
        };
        g.addEventListener('pointermove', show); g.addEventListener('focus', show);
        g.addEventListener('pointerleave', () => (tip.style.opacity = 0)); g.addEventListener('blur', () => (tip.style.opacity = 0));
        svg.append(g);
      });
      box.append(svg, tip);
    };
    draw();
    new ResizeObserver(() => draw()).observe(box);
    $('#daily').hidden = false;
  }

  function split(a) {
    const total = a.auto + a.manual || 1;
    const bar = $('#split-bar'); bar.replaceChildren();
    const s1 = el('span'); s1.style.width = (a.auto / total) * 100 + '%'; s1.style.background = 'var(--accent)';
    const s2 = el('span'); s2.style.width = (a.manual / total) * 100 + '%'; s2.style.background = 'var(--accent-2)';
    bar.append(s1, s2);
    bar.setAttribute('aria-label', `${a.auto} automatic, ${a.manual} by Chris`);
    $('#split-label').textContent = `${Math.round((a.auto / total) * 100)}% automatic · ${fmt.format(a.auto)} automatic, ${fmt.format(a.manual)} by Chris`;
    $('#automation').hidden = false;
  }

  function categories(rows) {
    const box = $('#category-bars'); box.replaceChildren();
    const max = Math.max(1, ...rows.map((r) => r.count));
    for (const r of rows) {
      const row = el('div', { class: 'bar-row' });
      const track = el('div', { class: 'bar-track' }); const fill = el('div', { class: 'bar-fill' }); fill.style.width = (r.count / max) * 100 + '%'; track.append(fill);
      row.append(el('span', {}, r.category), track, el('span', { class: 'num muted' }, fmt.format(r.count)));
      box.append(row);
    }
    $('#categories').hidden = false;
  }

  function companies(rows) {
    const box = $('#company-chips'); box.replaceChildren();
    for (const c of rows) { const chip = el('span', { class: 'chip', title: c.categories.join(', ') }, c.company); if (c.applications > 1) chip.append(el('b', {}, '×' + c.applications)); box.append(chip); }
    $('#company-count').textContent = `${fmt.format(rows.length)} companies`;
    $('#companies').hidden = false;
  }

  function pipeline(p) {
    for (const k of ['queued', 'applied', 'confirmed', 'interview', 'offer']) document.querySelector(`[data-p="${k}"]`).textContent = fmt.format(p[k]);
    $('#pipeline').hidden = false;
  }

  function requestForm() {
    $('#request').hidden = false;
    const form = $('#request-form'), status = $('#rq-status');
    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      const data = Object.fromEntries(new FormData(form));
      if (!data.name || !data.email || !data.reason) { status.className = 'status err'; status.textContent = 'Please fill in your name, email and what you would like to see.'; return; }
      if (DEMO) { status.className = 'status'; status.textContent = 'This is the demo, so nothing was sent. Use the form on the live page.'; return; }
      const btn = form.querySelector('button'); btn.disabled = true; status.className = 'status'; status.textContent = 'Sending…';
      try {
        const r = await fetch('/api/access-request', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
        const j = await r.json();
        if (!r.ok) throw new Error(j.error || 'Could not send.');
        form.reset(); status.className = 'status ok'; status.textContent = 'Request sent. Chris will get back to you by email.';
      } catch (err) { status.className = 'status err'; status.textContent = err.message; }
      btn.disabled = false;
    });
  }

  const DEMO = location.pathname === '/demo';
  if (DEMO) {
    $('#demo-banner').hidden = false; $('#demo-lede').hidden = false; $('#live-lede').hidden = true;
    document.title = 'Demo · Job Search HQ';
  }
  fetch(DEMO ? '/api/demo' : '/api/public').then((r) => r.json()).then((d) => {
    if (d.totals) totals(d.totals);
    if (d.daily && d.daily.length) dailyChart(d.daily);
    if (d.automation) split(d.automation);
    if (d.categories && d.categories.length) categories(d.categories);
    if (d.pipeline) pipeline(d.pipeline);
    if (d.companies && d.companies.length) companies(d.companies);
    if (d.request_form) requestForm();
  }).catch(() => {
    const p = el('p', { class: 'muted' }, 'The numbers could not load. Refresh to try again.');
    document.querySelector('header').append(p);
  });
})();
